## Station

```ts
interface Station extends Resource {
  type: "station";
  attributes: {
    name: string;
    seed: StationSeed;
    artworkUrl?: string;
    description?: string;
  };
  relationships?: {
    tracks?: List<Song>;
  };
}
```

## StationSeed

```ts
type StationSeed =
  | { kind: "resource"; type: "song" | "album" | "artist"; id: Id }
  | { kind: "genre"; name: string }
  | { kind: "popular" }
  | { kind: "custom" };
```
