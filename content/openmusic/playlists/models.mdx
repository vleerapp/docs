## Playlist

```ts
interface Playlist extends Resource {
  type: "playlist";
  attributes: {
    name: string;
    trackCount: number;
    curatorName?: string;
    artworkUrl?: string;
    description?: string;
  };
  relationships?: {
    tracks?: List<Song>;
  };
}
```
